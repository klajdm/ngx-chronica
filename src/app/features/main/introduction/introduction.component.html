NGX-Chronica

A comprehensive suite of 6 specialized date and time picker components for Angular. Fill critical gaps in the Angular ecosystem with production-ready components.

Why NGX-Chronica?

The Angular ecosystem lacks robust, production-ready Date & Time Picker components. NGX-Chronica addresses these critical gaps with a complete toolkit of date/time components that are production-ready, fully typed, and accessible.

NGX-Chronica provides 6 specialized components covering every date and time selection scenario-from simple date picking to complex duration selection. All components share a consistent API, theming system, and accessibility standards.

Quick Preview

./MyCalendar.component.ts
import { Component } from '@angular/core';
import { ChronicaInlineCalendarComponent } from 'ngx-chronica';

@Component({
  selector: 'app-my-calendar',
  standalone: true,
  imports: [ChronicaInlineCalendarComponent],
  template: `
    <chronica-inline-calendar
      [selectedDate]="selected"
      (dateSelected)="onDateSelected($event)"
    />
  `
})
export class MyCalendarComponent {
  selected: Date | null = new Date();

  onDateSelected(date: Date) {
    console.log('Selected:', date);
  }
}
